1a. NEW FILE `producers/claude_code/cc_agent.py` (stdlib only). Provide:

    AGENT_MAX = 64
    # Reserved producer-generated labels (a source value equal to one of these must never acquire its meaning).
    RESERVED = ("main", "custom", "unknown")
    # Evidence-backed built-in agent-type registry. The corpus (137 local sidecars) showed exactly these
    # built-in agentType values; the registry is intentionally minimal and extendable, and an unknown
    # built-in is treated as a custom name (allowlist-gated), never trusted automatically.
    BUILTIN = ("general-purpose", "fork")
    SAFE_AGENT_RE = re.compile(r"^(?:[A-Za-z][A-Za-z0-9_-]*|[A-Za-z][A-Za-z0-9_-]*:[A-Za-z][A-Za-z0-9_-]*)$")
    # The second alternative is the real plugin-agent shape: <plugin>:<agent> (observed as caveman:<agent>).

    def pseudonym(value: str) -> str        # "a-" + sha256(value).hexdigest()[:16]
    def sanitize(value, *, mode, allowlist, local_names) -> str
        # Returns one of: "main", "unknown", a BUILTIN member, an allowlisted safe custom name, or a
        # pseudonym "a-<hash>". Rules, in order:
        #   1. value not a non-empty str, or > AGENT_MAX -> "unknown"
        #   2. value in RESERVED (as a SOURCE value) -> pseudonym (never let a source spoof main/custom/unknown)
        #   3. value in BUILTIN -> value unchanged
        #   4. else it is a CUSTOM name: it may pass unchanged only when mode == "allowlist" AND
        #      SAFE_AGENT_RE matches AND value is an exact allowlist member AND value.lower() is not in
        #      {n.lower() for n in local_names} (detected local identifiers) -> value unchanged
        #   5. otherwise -> pseudonym(value)
        # Never trim, lower-case into a new value, slugify, extract a basename, decode, or salvage a
        # substring. A rejected value is NEVER returned and NEVER logged.
    def read_sidecar_agent_type(path: str) -> Optional[str]
        # For a subagent transcript file `<dir>/subagents/agent-<id>.jsonl`, read the sibling
        # `<dir>/subagents/agent-<id>.meta.json` and return its `agentType` when it is a non-empty str.
        # Read ONLY the `agentType` key. Never read `description`, `toolUseId`, `name` or any transcript.
        # Any error (missing/unreadable/oversized/invalid JSON) -> None. Bounded: refuse a file larger than
        # 64 KiB before parsing, and never raise.
